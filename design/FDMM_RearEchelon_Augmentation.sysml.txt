// ============================================================================
// FDMM_RearEchelon_Augmentation.sysml
// Complementary Rear-Echelon Augmentation Package for FDMM
// SysML v2 (text-style) single-file package with BFO grounding
//
// Purpose:
// - Define rear-echelon (fixed-site / higher-infrastructure) modules that augment
//   the forward-deployed container micro-factory.
// - Capture capabilities that are typically non-containerizable or operationally
//   unsuitable for the forward edge (foundation loads, precision environment,
//   safety envelope, power/utility intensity).
//
// BFO Grounding (high-level):
// - Modules/Equipment are BFO:MaterialEntity
// - Capabilities are BFO:RealizableEntity borne by Modules
// - Realizations are BFO:Process
// - Constraints are BFO:Quality / relational qualities (foundation, power, hazards)
//
// Source grounding: Equipment items correspond to the provided equipment list.
// ============================================================================

package FDMM_RearEchelon_Augmentation {

  // --------------------------------------------------------------------------
  // 00. Minimal shared terminology (keep file standalone)
  // --------------------------------------------------------------------------
  package _BFO_Terminology {

    annotation bfoType { attribute iri: String; }
    annotation bfoNote { attribute text: String; }

  }

  package Context {

    value type kW { unit = "kW"; }
    value type scfm { unit = "scfm"; }
    value type psi { unit = "psi"; }
    value type kg { unit = "kg"; }
    value type degC { unit = "degC"; }
    value type percent { unit = "%"; }

    enum EnvironmentalControlClass { None, Basic, Conditioned, Precision }
    enum PowerQualityClass { Dirty, Conditioned, Precision }
    enum HandlingClass { Manual, Forklift, Crane, RiggingTeam }

    part def RearSite {
      doc /*BFO: MaterialEntity (site proxy for rear-echelon base)*/;
      attribute ambientTemp_degC: degC;
      attribute ambientRH_percent: percent;
      attribute availablePowerQuality: PowerQualityClass;
      attribute availablePower_kW: kW;
      attribute availableAir_scfm: scfm;
      attribute availableAir_psi: psi;
      attribute dustControl: EnvironmentalControlClass;
      attribute craneAvailable: Boolean;
      attribute riggingAvailable: Boolean;
      attribute fireSuppressionClass: String;
      attribute wastewaterHandling: Boolean;
    }

  }

  package Interfaces {
    import Context::*;

    interface def ElectricalPower {
      attribute max_kW: kW;
      attribute voltage: String;
      attribute phase: String;
      attribute frequency: String;
      attribute quality: PowerQualityClass;
    }

    interface def CompressedAir {
      attribute pressure_psi: psi;
      attribute flow_scfm: scfm;
      attribute drynessClass: String;
    }

    // Rear-echelon introduces new logistics interfaces
    interface def WorkOrder {
      attribute workOrderId: String;
      attribute priority: String;     // e.g., P1/P2/P3
      attribute partNumber: String;
      attribute requiredBy: String;   // date/time string for planning
      attribute notes: String;
    }

    interface def FinishedGoods {
      attribute partNumber: String;
      attribute qty: Integer;
      attribute packaging: String;
      attribute shippingClass: String;
    }

    interface def QualityRecord {
      attribute recordId: String;
      attribute inspectionLevel: String;
      attribute traceability: String;
    }

  }

  // --------------------------------------------------------------------------
  // 01. Capabilities (rear-echelon set)
  // --------------------------------------------------------------------------
  package Capabilities {
    import _BFO_Terminology::*;

    class Capability {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000016" } // realizable entity
      @bfoNote { text = "Rear-echelon capability modeled as BFO RealizableEntity borne by rear modules." }
      attribute capabilityId: String;
      attribute description: String;
      attribute forwardDependency: String; // how it supports forward ops
    }

    // Rear-echelon unique/expanded capabilities
    class Cap_LargeEnvelopeTurning : Capability;
    class Cap_HeavyStampingPunching : Capability;
    class Cap_WireEDM_Precision : Capability;
    class Cap_RotaryGrinding_Precision : Capability;
    class Cap_AdvancedHeatTreat : Capability;
    class Cap_ProductionTurretTurning : Capability;

    // Interface/coordination capabilities
    class Cap_WorkOrderIntakeAndDispatch : Capability;
    class Cap_QualityAssuranceAndTrace : Capability;

  }

  // --------------------------------------------------------------------------
  // 02. Physical architecture (rear-echelon modules & equipment)
  // --------------------------------------------------------------------------
  package PhysicalArchitecture {
    import _BFO_Terminology::*;
    import Context::*;
    import Interfaces::*;
    import Capabilities::*;

    // Equipment base class
    part def Equipment {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material entity
      attribute equipId: String;
      attribute makeModel: String;
      attribute mass_kg: kg;
      attribute handling: HandlingClass;

      // Deployment-relevant qualities
      attribute requiresFoundation: Boolean;
      attribute requiresVibrationIsolation: Boolean;
      attribute requiresEnvironmentalControl: EnvironmentalControlClass;
      attribute requiresConditionedPower: Boolean;
      attribute hazardClass: String;
      attribute utilityNotes: String;
    }

    // Rear-specific equipment types
    part def LargeManualLathe : Equipment;
    part def TurretLathe : Equipment;
    part def PunchPress : Equipment;
    part def WireEDM : Equipment;
    part def RotaryGrinder : Equipment;
    part def HeatTreatFurnace : Equipment;
    part def AirCompressor : Equipment;

    // Rear module abstraction
    part def Rear_Module {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material entity
      attribute moduleId: String;
      attribute fixedSiteFootprint: String; // e.g., "bay", "cell", "room"
      attribute grossMass_kg: kg;

      attribute powerDemand_kW: kW;
      attribute airDemand_scfm: scfm;
      attribute thermalLoad_kW: kW;
      attribute envControl: EnvironmentalControlClass;

      port pwrIn : ElectricalPower;
      port airIn : CompressedAir;

      // Rear ops I/O
      port woIn : WorkOrder;
      port goodsOut : FinishedGoods;
      port qaOut : QualityRecord;

      part capabilities[*] : Capability;

      attribute forwardSupportMode: String; // "build-to-order", "repair-return", "batch-stock", etc.
    }

    // Rear modules (augmenters)
    part def Module_LargeTurning : Rear_Module {
      part lathes[*] : LargeManualLathe;
      part turret[*] : TurretLathe;

      part cap_large_turn : Cap_LargeEnvelopeTurning;
      part cap_turret_prod : Cap_ProductionTurretTurning;
    }

    part def Module_WireEDM : Rear_Module {
      part edm[*] : WireEDM;

      part cap_edm : Cap_WireEDM_Precision;
    }

    part def Module_HeavyGrinding : Rear_Module {
      part rotary[*] : RotaryGrinder;

      part cap_rotary : Cap_RotaryGrinding_Precision;
    }

    part def Module_StampingPunching : Rear_Module {
      part press[*] : PunchPress;

      part cap_punch : Cap_HeavyStampingPunching;
    }

    part def Module_HeatTreat_Advanced : Rear_Module {
      part furnaces[*] : HeatTreatFurnace;

      part cap_ht : Cap_AdvancedHeatTreat;
    }

    part def Module_QA_Dispatch : Rear_Module {
      part cap_dispatch : Cap_WorkOrderIntakeAndDispatch;
      part cap_qa : Cap_QualityAssuranceAndTrace;
    }

    // Rear site utilities
    part def Rear_Utilities {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" }
      part compressors[*] : AirCompressor;
      port pwrOut : ElectricalPower;
      port airOut : CompressedAir;
      attribute notes: String;
    }

    // Top-level rear-echelon augmentation site
    part def RearEchelonFacility {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" }
      part site : Context::RearSite;
      part utilities : Rear_Utilities;
      part modules[*] : Rear_Module;

      attribute facilityName: String;
      attribute conceptOfOps: String;
    }

  }

  // --------------------------------------------------------------------------
  // 03. Functional architecture (processes / realizations)
  // --------------------------------------------------------------------------
  package FunctionalArchitecture {
    import _BFO_Terminology::*;

    activity def Process {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000015" } // process
      attribute processId: String;
      attribute description: String;
    }

    // Rear processes
    activity def P_LargeEnvelopeTurning : Process;
    activity def P_TurretLatheProduction : Process;
    activity def P_WireEDM : Process;
    activity def P_RotaryGrinding : Process;
    activity def P_StampingPunching : Process;
    activity def P_AdvancedHeatTreat : Process;

    activity def P_WorkOrderTriage : Process;
    activity def P_QA_InspectionAndTrace : Process;
    activity def P_PackAndShipForward : Process;

  }
