// ============================================================================
// FDMM_BFO.sysml
// Forward-Deployed Modular Manufacturing (FDMM)
// SysML v2 (text-style) single-file package with BFO grounding
//
// Notes:
// - This file is written in a SysMLv2-text *style* intended for planning/model seeding.
// - Tools vary in how strictly they enforce current SysMLv2 grammar. If your tool
//   requires adjustments, treat this as an authoritative conceptual baseline.
//
// BFO Grounding (high-level):
// - Modules/Equipment/Utilities/People are BFO:MaterialEntity (independent continuants)
// - Capabilities are BFO:RealizableEntity (Functions/Dispositions) borne by Modules
// - Realizations are BFO:Process (occurrents) expressed as Activities/UseCases
// - Constraints are BFO:Quality / relational quality expressed as value properties
//
// Source grounding: Equipment items correspond to the provided equipment list.
// ============================================================================

package FDMM_SystemOfInterest {

  // --------------------------------------------------------------------------
  // 00. Profile-ish definitions (lightweight)
  // --------------------------------------------------------------------------
  package _BFO_Terminology {

    // Abstract BFO categories (documentation-only; used via comments/metadata)
    abstract class BFO_Entity;
    abstract class BFO_Continuant : BFO_Entity;
    abstract class BFO_Occurrent  : BFO_Entity;

    abstract class BFO_IndependentContinuant : BFO_Continuant;   // MaterialEntity, etc.
    abstract class BFO_SpecificallyDependentContinuant : BFO_Continuant; // realizables, qualities
    abstract class BFO_RealizableEntity : BFO_SpecificallyDependentContinuant; // functions, roles
    abstract class BFO_Quality : BFO_SpecificallyDependentContinuant;

    abstract class BFO_Process : BFO_Occurrent;

    // "Tags" used as doc annotations (keep simple; tool-friendly)
    annotation bfoType { attribute iri: String; }
    annotation bfoNote { attribute text: String; }

  }

  // --------------------------------------------------------------------------
  // 01. Context & assumptions (planning baseline)
  // --------------------------------------------------------------------------
  package Context {

    enum ContainerType { ISO_20ft, ISO_40ft, ISO_40ft_HC }

    // Value types
    value type kW { unit = "kW"; }
    value type scfm { unit = "scfm"; }
    value type psi { unit = "psi"; }
    value type kg { unit = "kg"; }
    value type m { unit = "m"; }
    value type degC { unit = "degC"; }
    value type percent { unit = "%"; }
    value type mm { unit = "mm"; }

    // Qualitative classes for planning (can be refined to numeric later)
    enum EnvironmentalControlClass { None, Basic, Conditioned }
    enum PowerQualityClass { Dirty, Conditioned }
    enum HandlingClass { Manual, Forklift, Crane }

    // Deployment "site" is a context object that supplies constraints
    part def DeploymentSite {
      doc /*BFO: MaterialEntity (site as material aggregate / location proxy)*/;
      attribute ambientTemp_degC: degC;
      attribute ambientRH_percent: percent;
      attribute availablePowerQuality: PowerQualityClass;
      attribute availablePower_kW: kW;
      attribute availableAir_scfm: scfm;
      attribute availableAir_psi: psi;
      attribute dustLevel: EnvironmentalControlClass;
      attribute groundStabilityClass: String;
    }

  }

  // --------------------------------------------------------------------------
  // 02. Interfaces (ports/item flows)
  // --------------------------------------------------------------------------
  package Interfaces {
    import Context::*;

    interface def ElectricalPower {
      attribute max_kW: kW;
      attribute voltage: String;
      attribute phase: String;
      attribute frequency: String;
      attribute quality: PowerQualityClass;
    }

    interface def CompressedAir {
      attribute pressure_psi: psi;
      attribute flow_scfm: scfm;
      attribute drynessClass: String;
    }

    interface def DataNet {
      attribute bandwidth: String;
      attribute latency: String;
      attribute offlineMode: Boolean;
    }

    interface def MaterialFlow {
      attribute form: String;           // bar, plate, billet, consumables, etc.
      attribute handling: HandlingClass;
      attribute mass_kg: kg;
    }

    interface def HeatRejection {
      attribute thermalLoad_kW: kW;
      attribute exhaustType: String; // filtered/vented/recirculated
    }
  }

  // --------------------------------------------------------------------------
  // 03. BFO-grounded capabilities (Realizable entities borne by modules)
  // --------------------------------------------------------------------------
  package Capabilities {
    import _BFO_Terminology::*;
    import Context::*;

    // Abstract capability concept (BFO:Function / Disposition)
    class Capability {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000016" } // realizable entity
      @bfoNote { text = "Capability modeled as a BFO RealizableEntity (Function/Disposition) borne by a module." }
      attribute capabilityId: String;
      attribute description: String;
      attribute containerSuitability: String; // ContainerCompatible / Adaptable / RearEchelon
    }

    // Upper-level taxonomy
    class SustainmentRepairCapability : Capability;
    class LowRateProductionCapability : Capability;
    class MetrologyProcessAssuranceCapability : Capability;
    class DeploymentLogisticsSupportCapability : Capability;

    // Leaf capabilities (planning set)
    class Cap_CutAndPrepStock : SustainmentRepairCapability;
    class Cap_JoinAndFabricate : SustainmentRepairCapability;
    class Cap_MachineRotational_Manual : SustainmentRepairCapability;
    class Cap_MachinePrismatic_Manual  : SustainmentRepairCapability;
    class Cap_MachineRotational_CNC    : LowRateProductionCapability;
    class Cap_MachinePrismatic_CNC     : LowRateProductionCapability;
    class Cap_FinishAndDeburr          : SustainmentRepairCapability;
    class Cap_RestoreFits_HoneGrind     : SustainmentRepairCapability;
    class Cap_HeatTreat_Limited         : SustainmentRepairCapability;
    class Cap_DesignPlanProgram_CADCAM  : MetrologyProcessAssuranceCapability;
    class Cap_HandleMaterials_Yard      : DeploymentLogisticsSupportCapability;

  }

  // --------------------------------------------------------------------------
  // 04. Personnel & roles (BFO roles realized in processes)

      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material entity
      attribute name: String;
    }

    class Role {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000023" } // role (realizable)
      attribute roleName: String;
    }

    // Roles typically required
    class MachinistRole : Role;
    class CNCProgrammerRole : Role;
    class WelderRole : Role;
    class FabricatorRole : Role;
    class MaintainerRole : Role;

      part cncProgrammers[*] : Person;
      part welders[*] : Person;
      part logistics[*] : Person;

      // Informal role assignment (can be refined to allocations)
      attribute notes: String;
    }
  }

  // --------------------------------------------------------------------------
  // 05. Physical architecture (modules, equipment, utilities)
  // --------------------------------------------------------------------------
  package PhysicalArchitecture {
    import _BFO_Terminology::*;
    import Context::*;
    import Interfaces::*;
    import Capabilities::*;

    // ----------------------------
    // Equipment abstractions
    // ----------------------------
    part def Equipment {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material entity
      attribute equipId: String;
      attribute makeModel: String;
      attribute mass_kg: kg;
      attribute handling: HandlingClass;

      // Deployment-relevant qualities (BFO:Qualities)
      attribute requiresFoundation: Boolean;
      attribute requiresVibrationIsolation: Boolean;
      attribute requiresEnvironmentalControl: EnvironmentalControlClass;
      attribute requiresConditionedPower: Boolean;
      attribute hazardClass: String;
    }

    part def ManualMill     : Equipment;
    part def ManualLathe    : Equipment;
    part def DrillPress     : Equipment;
    part def SurfaceGrinder : Equipment;
    part def HoningMachine  : Equipment;
    part def CNC_VMC        : Equipment;
    part def CNC_Lathe      : Equipment;
    part def Saw            : Equipment;
    part def PlasmaCutter   : Equipment;
    part def WeldingSystem  : Equipment;
    part def StudWelder     : Equipment;
    part def DeburrMachine  : Equipment;
    part def HeatTreatFurnace : Equipment;
    part def ComputeNode    : Equipment;
    part def AirCompressor  : Equipment;
    part def Forklift       : Equipment;
    part def FlatbedTruck   : Equipment;

    // ----------------------------
    // Container module abstraction
    // ----------------------------
    part def Container_Module {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // module is material entity
      attribute moduleId: String;
      attribute containerType: ContainerType;
      attribute grossMass_kg: kg;

      // Resource demands (planning)
      attribute powerDemand_kW: kW;
      attribute airDemand_scfm: scfm;
      attribute thermalLoad_kW: kW;
      attribute envControl: EnvironmentalControlClass;

      // Ports (interfaces)
      port pwrIn : ElectricalPower;
      port airIn : CompressedAir;
      port netIn : DataNet;
      port heatOut : HeatRejection;
      port matIn : MaterialFlow;
      port matOut: MaterialFlow;

      // BFO-realizable capabilities borne by the module
      part capabilities[*] : Capability;

      // Deployment suitability notes
      attribute containerCompatibility: String; // Compatible / Adaptable / RearEchelon
    }

    // ----------------------------
    // Module specializations (cells)
    // ----------------------------
    part def Module_ManualMachining : Container_Module {
      part mills[*] : ManualMill;
      part lathes[*] : ManualLathe;
      part drills[*] : DrillPress;
      part grinders[*] : SurfaceGrinder;
      part hone[*] : HoningMachine;

      // Typical capability-bearing set
      part cap_mill : Cap_MachinePrismatic_Manual;
      part cap_turn : Cap_MachineRotational_Manual;
      part cap_restore : Cap_RestoreFits_HoneGrind;
    }

    part def Module_CNCMachining : Container_Module {
      part vmc[*] : CNC_VMC;
      part cncLathes[*] : CNC_Lathe;
      part cadcam[*] : ComputeNode;

      part cap_cnc_mill : Cap_MachinePrismatic_CNC;
      part cap_cnc_turn : Cap_MachineRotational_CNC;
      part cap_cadcam : Cap_DesignPlanProgram_CADCAM;
    }

    part def Module_CuttingPrep : Container_Module {
      part saws[*] : Saw;
      part plasma[*] : PlasmaCutter;

      part cap_cut : Cap_CutAndPrepStock;
    }

    part def Module_WeldingFab : Container_Module {
      part welding[*] : WeldingSystem;
      part stud[*] : StudWelder;

      part cap_join : Cap_JoinAndFabricate;
    }

    part def Module_Finishing : Container_Module {
      part deburr[*] : DeburrMachine;
      part grinders[*] : SurfaceGrinder;

      part cap_finish : Cap_FinishAndDeburr;
    }

    part def Module_HeatTreat : Container_Module {
      part furnaces[*] : HeatTreatFurnace;
      part cap_ht : Cap_HeatTreat_Limited;
    }

    // ----------------------------
    // Yard operations (non-container)
    // ----------------------------
    part def Yard_Operations {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material aggregate
      part forklifts[*] : Forklift;
      part flatbeds[*] : FlatbedTruck;

      part cap_handle : Cap_HandleMaterials_Yard;
    }

    // ----------------------------
    // Utilities (some implied by list; compressors present)
    // ----------------------------
    part def Utilities {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" }
      part compressors[*] : AirCompressor;

      // Ports outward to modules (modeled as provided services)
      port pwrOut : ElectricalPower;
      port airOut : CompressedAir;
      port netOut : DataNet;

      attribute notes: String;
    }

    // ----------------------------
    // Top-level deployment (system-of-systems)
    // ----------------------------
    part def FDMM_Deployment {
      @bfoType { iri = "http://purl.obolibrary.org/obo/BFO_0000040" } // material entity aggregate
      part site : Context::DeploymentSite;

      part utilities : Utilities;
      part yard : Yard_Operations;
      part modules[*] : Container_Module;

      attribute deploymentName: String;
      attribute conceptOfOps: String;
    }

  }
